import java.awt.*;
import java.awt.image.BufferedImage;
import java.io.File;
import javax.imageio.ImageIO;

/** Round 2: plain number and glyph both in the middle of the oval rune. */
public class Mock4 {
	static final int[][] KIND = {
		{0x2a1012, 0xa8585e, 0x803c44, 0x56262c}, {0x10182c, 0x5a78a8, 0x3e5684, 0x283a5c},
		{0x122a14, 0x5e9a5e, 0x427442, 0x2a502c}, {0x0e2626, 0x529690, 0x3a6e6a, 0x264a48}};
	static final int GLYPH = 0xff9a2e, GLYPH_HI = 0xffd27a, NUM = 0xfff4d0, SHADOW = 0x1a0e08;
	static final String[][] G7 = {
		{"...#...", "..###..", "...#...", ".#.#.#.", "..###..", "...#..."},
		{".#####.", ".#...#.", ".#.#.#.", ".#.#.#.", "..#.#..", "...#..."},
		{".#.#.#.", ".#.#.#.", ".#####.", "...#...", "...#...", "...#..."},
		{".#####.", "#..#..#", "...#...", "...#...", "...#...", "...#..."}};
	static final String[][] G5 = {
		{"..#..", ".###.", "..#..", "#.#.#", ".###."}, {"#####", "#.#.#", "#.#.#", ".#.#.", "..#.."},
		{"#.#.#", "#.#.#", "#####", "..#..", "..#.."}, {"#####", "#.#.#", "..#..", "..#..", "..#.."}};
	static final String[] D35 = {"###,#.#,#.#,#.#,###", ".#.,##.,.#.,.#.,###", "###,..#,###,#..,###", "###,..#,###,..#,###",
		"#.#,#.#,###,..#,..#", "###,#..,###,..#,###", "###,#..,###,#.#,###", "###,..#,..#,.#.,.#.", "###,#.#,###,#.#,###", "###,#.#,###,..#,###"};
	// game-font-like digits, 5x7
	static final String[] D57 = {
		".###.,#...#,#..##,#.#.#,##..#,#...#,.###.", "..#..,.##..,..#..,..#..,..#..,..#..,#####", ".###.,#...#,....#,..##.,.#...,#....,#####",
		".###.,#...#,....#,..##.,....#,#...#,.###.", "...#.,..##.,.#.#.,#..#.,#####,...#.,...#.", "#####,#....,####.,....#,....#,#...#,.###.",
		"..##.,.#...,#....,####.,#...#,#...#,.###.", "#####,#...#,....#,...#.,..#..,..#..,..#..", ".###.,#...#,#...#,.###.,#...#,#...#,.###.",
		".###.,#...#,#...#,.####,....#,...#.,.##.."};
	static final String[] NAMES = {"Sharpness", "Protection", "Throw Power", "Efficiency"};
	static final int[] LEVELS = {1, 3, 8, 12};

	interface Mask { boolean in(int x, int y); }
	static int[][] px;
	static void set(int x, int y, int c) { if (x >= 0 && y >= 0 && x < 16 && y < 16) px[y][x] = 0xFF000000 | c; }
	static double sq(double v) { return v * v; }
	static final Mask OVAL = (x, y) -> x >= 1 && x <= 13 && y >= 0 && y <= 15 && (y <= 7 || sq((x - 7) / 6.6) + sq((y - 7) / 8.6) <= 1);

	static void stone(int[] pal) {
		Mask m = OVAL;
		for (int y = 0; y < 16; y++) for (int x = 0; x < 16; x++) {
			if (!m.in(x, y)) continue;
			boolean edge = !m.in(x - 1, y) || !m.in(x + 1, y) || !m.in(x, y - 1) || !m.in(x, y + 1);
			int c = pal[2];
			if (edge) c = pal[0];
			else if (!m.in(x - 2, y) || !m.in(x, y - 2)) c = pal[1];
			else if (!m.in(x + 2, y) || !m.in(x, y + 2)) c = pal[3];
			else if (((x * 7 + y * 13) % 11) == 0) c = pal[3];
			set(x, y, c);
		}
	}

	static void bits(String[] rows, int ox, int oy, int c, boolean hi) {
		for (int y = 0; y < rows.length; y++) for (int x = 0; x < rows[y].length(); x++)
			if (rows[y].charAt(x) == '#') set(ox + x, oy + y, hi && (x + y) % 5 == 1 ? GLYPH_HI : c);
	}

	/** A number centred on cx with its top at y; font 35 (3x5) or 57 (5x7); shadow one pixel down-right. */
	static void number(int n, int cx, int y, boolean big, boolean shadow) {
		String s = String.valueOf(n);
		int w = big ? 5 : 3, total = s.length() * (w + 1) - 1, x = cx - (total - 1) / 2;
		for (int pass = shadow ? 0 : 1; pass < 2; pass++) {
			int xx = x;
			for (char ch : s.toCharArray()) {
				String[] rows = (big ? D57 : D35)[ch - '0'].split(",");
				bits(rows, xx + (pass == 0 ? 1 : 0), y + (pass == 0 ? 1 : 0), pass == 0 ? SHADOW : NUM, false);
				xx += w + 1;
			}
		}
	}

	static void dim(String[] rows, int ox, int oy, int[] pal) { bits(rows, ox, oy, pal[3], false); }

	static int[][] icon(int v, int k, int level) {
		px = new int[16][16];
		int[] pal = KIND[k];
		stone(pal);
		switch (v) {
			case 5 -> { bits(G7[k], 4, 2, GLYPH, true); number(level, 7, 9, false, false); }
			case 1 -> { bits(G7[k], 4, 1, GLYPH, true); number(level, 8, 9, false, true); }       // glyph above, small number below
			case 2 -> { bits(G5[k], 5, 1, GLYPH, true); number(level, 8, 7, true, true); }        // small glyph above, big number below
			case 3 -> { dim(G7[k], 4, 4, pal); bits(G7[k], 4, 4, GLYPH, true); for (int y = 6; y <= 10; y++) for (int x = 4; x <= 11; x++) if (px[y][x] != 0) set(x, y, pal[3]);
				number(level, 8, 6, false, true); }                                                     // glyph centred, number on a dark plate over its middle
			default -> { dim(G7[k], 4, 4, new int[]{0, 0, 0, darker(pal[2])}); number(level, 8, 4, true, true); } // engraved glyph behind, big number in front
		}
		return px;
	}

	static int darker(int c) {
		int r = (c >> 16) & 255, g = (c >> 8) & 255, b = c & 255;
		return ((r * 3 / 5) << 16) | ((g * 3 / 5) << 8) | (b * 3 / 5);
	}

	static final String[] TITLES = {"", "A. Glyph above, small number below", "B. Small glyph above, big number below",
		"C. Glyph centred, number on a plate across it", "D. Engraved glyph behind, big number in front", "A3. Glyph 1 px lower, a free row above the number"};

	static void blit(Graphics2D g, int[][] p, int x0, int y0, int s) {
		for (int y = 0; y < 16; y++) for (int x = 0; x < 16; x++) if (p[y][x] != 0) { g.setColor(new Color(p[y][x], true)); g.fillRect(x0 + x * s, y0 + y * s, s, s); }
	}

	static void slot(Graphics2D g, int x, int y, int s) {
		g.setColor(new Color(0x8b8b8b)); g.fillRect(x, y, s, s);
		g.setColor(new Color(0x373737)); g.fillRect(x, y, s, 2); g.fillRect(x, y, 2, s);
		g.setColor(Color.WHITE); g.fillRect(x, y + s - 2, s, 2); g.fillRect(x + s - 2, y, 2, s);
	}

	public static void main(String[] a) throws Exception {
		int big = 8, pw = 4 * (16 * big + 20) + 20, ph = 40 + 16 * big + 26 + 60;
		BufferedImage sheet = new BufferedImage(pw * 2, ph * 2, BufferedImage.TYPE_INT_ARGB);
		Graphics2D sg = sheet.createGraphics();
		for (int v = 5; v <= 5; v++) {
			BufferedImage panel = new BufferedImage(pw, ph, BufferedImage.TYPE_INT_ARGB);
			Graphics2D g = panel.createGraphics();
			g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
			g.setColor(new Color(0x2a2a32)); g.fillRect(0, 0, pw, ph);
			g.setColor(new Color(0x9a7a20)); g.drawRect(0, 0, pw - 1, ph - 1);
			g.setColor(Color.WHITE); g.setFont(new Font("SansSerif", Font.BOLD, 20)); g.drawString(TITLES[v], 14, 28);
			for (int k = 0; k < 4; k++) {
				int[][] p = icon(v, k, LEVELS[k]);
				int x = 20 + k * (16 * big + 20), y = 40;
				blit(g, p, x, y, big);
				g.setColor(new Color(0xc8c8d0)); g.setFont(new Font("SansSerif", Font.PLAIN, 14));
				g.drawString(NAMES[k] + " " + LEVELS[k], x, y + 16 * big + 18);
				int sy = y + 16 * big + 28;
				slot(g, x, sy, 36); blit(g, p, x + 2, sy + 2, 2);
				slot(g, x + 44, sy, 52); blit(g, p, x + 46, sy + 2, 3);
			}
			g.dispose();
			ImageIO.write(panel, "png", new File(a[0], "centerA3.png"));
			
		}
		
	}
}
